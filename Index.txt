  <script>
    // State Application
    let state = {
      transactions: [],
      categories: [],
      summary: { totalSaldo: 0, totalMasuk: 0, totalKeluar: 0 },
      page: 1,
      limit: 15,
      hasMore: false,
      search: '',
      type: 'ALL',
      startDate: '',
      endDate: '',
      pendingDeleteId: null,
      searchTimer: null
    };

    // Initial Load Handler
    document.addEventListener('DOMContentLoaded', function() {
      // Set Default Date to Today in Form
      const today = new Date().toISOString().split('T')[0];
      document.getElementById('formTanggal').value = today;

      // Safe Apps Script Runner Wrapper
      if (typeof google !== 'undefined' && google.script && google.script.run) {
        fetchInitialData();
      } else {
        // Fallback Mock Data for Gemini Canvas Safe Preview Mode
        loadMockData();
      }
    });

    function fetchInitialData() {
      showToast('Memuat data IuranKU...', 'info');
      
      // Get Categories
      google.script.run
        .withSuccessHandler(function(cats) {
          state.categories = cats || [];
          populateCategoryDropdown();
        })
        .getCategories();

      // Get Summary
      refreshSummary();

      // Get Transactions Page 1
      state.page = 1;
      fetchTransactions(false);
    }

    function refreshSummary() {
      if (typeof google === 'undefined') return;
      google.script.run
        .withSuccessHandler(function(summary) {
          if (summary && !summary.error) {
            state.summary = summary;
            renderSummary();
          }
        })
        .getDashboardSummary();
    }

    function fetchTransactions(append) {
      if (typeof google === 'undefined') return;

      const params = {
        search: state.search,
        type: state.type,
        startDate: state.startDate,
        endDate: state.endDate,
        page: state.page,
        limit: state.limit
      };

      google.script.run
        .withSuccessHandler(function(res) {
          if (res && !res.error) {
            if (append) {
              state.transactions = state.transactions.concat(res.transactions);
            } else {
              state.transactions = res.transactions;
            }
            state.hasMore = res.hasMore;
            renderTransactions();
            document.getElementById('recordCountBadge').innerText = res.total + ' Data';
          }
        })
        .getTransactions(params);
    }

    // SAFE DOM RENDERING - String Concatenation without Nested Backticks
    function renderTransactions() {
      const tbody = document.getElementById('transactionTableBody');
      const emptyState = document.getElementById('emptyState');
      const loadMoreContainer = document.getElementById('loadMoreContainer');

      if (!state.transactions || state.transactions.length === 0) {
        tbody.innerHTML = '';
        emptyState.classList.remove('hidden');
        loadMoreContainer.classList.add('hidden');
        return;
      }

      emptyState.classList.add('hidden');
      let html = '';

      for (let i = 0; i < state.transactions.length; i++) {
        const item = state.transactions[i];
        const isMasuk = item.jenis === 'Masuk';
        const jenisBadge = isMasuk 
          ? '<span class="inline-flex items-center px-2.5 py-1 rounded-full text-[10px] font-bold bg-emerald-500/10 text-emerald-400 border border-emerald-500/30">Masuk</span>'
          : '<span class="inline-flex items-center px-2.5 py-1 rounded-full text-[10px] font-bold bg-red-500/10 text-dangerred border border-red-500/30">Keluar</span>';

        const nominalFormatted = (isMasuk ? '+ ' : '- ') + formatRupiah(item.nominal);
        const nominalColor = isMasuk ? 'text-emerald-400 font-bold' : 'text-dangerred font-bold';

        // Classic String Concatenation
        html += '<tr class="hover:bg-slate-900/60 transition duration-150">' +
                  '<td class="p-3.5 whitespace-nowrap">' +
                    '<div class="font-mono text-xs font-semibold text-neon">' + escapeHtml(item.id) + '</div>' +
                    '<div class="text-[11px] text-slate-400 mt-0.5">' + escapeHtml(item.tanggal) + '</div>' +

    function renderSummary() {
      document.getElementById('statTotalSaldo').innerText = formatRupiah(state.summary.totalSaldo);
      document.getElementById('statTotalMasuk').innerText = formatRupiah(state.summary.totalMasuk);
      document.getElementById('statTotalKeluar').innerText = formatRupiah(state.summary.totalKeluar);
    }

    function populateCategoryDropdown() {
      const select = document.getElementById('formKategori');
      const currentJenis = document.getElementById('formJenis').value;
      select.innerHTML = '';

      const filtered = state.categories.filter(function(c) {
        return c.type === currentJenis;
      });

      if (filtered.length === 0) {
        const defaultOption = document.createElement('option');
        defaultOption.value = 'Umum';
        defaultOption.innerText = 'Umum';
        select.appendChild(defaultOption);
        return;
      }

      for (let i = 0; i < filtered.length; i++) {
        const opt = document.createElement('option');
        opt.value = filtered[i].name;
        opt.innerText = filtered[i].name;
        select.appendChild(opt);
      }
    }

    function handleFormSubmit(e) {
      e.preventDefault();

      const txId = document.getElementById('formTxId').value;
      const jenis = document.getElementById('formJenis').value;
      const tanggalInput = document.getElementById('formTanggal').value;
      const kategori = document.getElementById('formKategori').value;
      const nominal = parseFloat(document.getElementById('formNominal').value) || 0;
      const deskripsi = document.getElementById('formDeskripsi').value;

      if (!nominal || nominal <= 0) {
        showToast('Nominal harus lebih dari 0', 'error');
        return;
      }

      // Convert date to DD/MM/YYYY format
      const dateParts = tanggalInput.split('-');
      const formattedDate = dateParts.length === 3 ? dateParts[2] + '/' + dateParts[1] + '/' + dateParts[0] : tanggalInput;

      const payload = {
        id: txId,
        jenis: jenis,
        tanggal: formattedDate,
        kategori: kategori,
        nominal: nominal,
        deskripsi: deskripsi
      };

      /* OPTIMISTIC UI UPDATE */
      if (!txId) {
        const tempObj = {
          id: 'TRX-TEMP',
          tanggal: formattedDate,
          jenis: jenis,
          kategori: kategori,
          nominal: nominal,
          deskripsi: deskripsi
        };
        state.transactions.unshift(tempObj);
        if (jenis === 'Masuk') {
          state.summary.totalMasuk += nominal;
        } else {
          state.summary.totalKeluar += nominal;
        }
        state.summary.totalSaldo = state.summary.totalMasuk - state.summary.totalKeluar;
        renderSummary();
        renderTransactions();
      }

      closeModal();
      showToast('Menyimpan data transaksi...', 'info');

      if (typeof google !== 'undefined' && google.script && google.script.run) {
        google.script.run
          .withSuccessHandler(function(res) {
            if (res && res.success) {
              showToast(res.message, 'success');
              refreshSummary();
              state.page = 1;
              fetchTransactions(false);
            } else {
              showToast(res ? res.message : 'Gagal menyimpan transaksi.', 'error');
            }
          })
          .saveTransaction(payload);
      } else {
        showToast('Mode Standalone Preview: Transaksi tersimpan lokal', 'success');
      }
    }

    function confirmDelete(id) {
      state.pendingDeleteId = id;
      document.getElementById('deleteModal').classList.remove('hidden');
    }

    function closeDeleteModal() {
      state.pendingDeleteId = null;
      document.getElementById('deleteModal').classList.add('hidden');
    }

    function executeDelete() {
      const id = state.pendingDeleteId;
      if (!id) return;

      closeDeleteModal();

      // OPTIMISTIC UI REMOVAL
      const idx = state.transactions.findIndex(function(t) { return t.id === id; });
      if (idx !== -1) {
        const deletedItem = state.transactions[idx];
        if (deletedItem.jenis === 'Masuk') {
          state.summary.totalMasuk -= deletedItem.nominal;
        } else {
          state.summary.totalKeluar -= deletedItem.nominal;
        }
        state.summary.totalSaldo = state.summary.totalMasuk - state.summary.totalKeluar;
        state.transactions.splice(idx, 1);
        renderSummary();
        renderTransactions();
      }

      showToast('Menghapus transaksi ' + id + '...', 'info');

      if (typeof google !== 'undefined' && google.script && google.script.run) {
        google.script.run
          .withSuccessHandler(function(res) {
            if (res && res.success) {
              showToast(res.message, 'success');
              refreshSummary();
              state.page = 1;
              fetchTransactions(false);
            } else {
              showToast(res ? res.message : 'Gagal menghapus transaksi.', 'error');
            }
          })
          .deleteTransaction(id);
      } else {
        showToast('Mode Standalone Preview: Item berhasil dihapus', 'success');
      }
    }

    function openModal(defaultType) {
      document.getElementById('formTxId').value = '';
      document.getElementById('modalTitle').innerText = defaultType === 'Masuk' ? 'Input Dana Masuk' : 'Input Dana Keluar';
      document.getElementById('formNominal').value = '';
      document.getElementById('formDeskripsi').value = '';
      selectTypeInForm(defaultType || 'Masuk');
      document.getElementById('transactionModal').classList.remove('hidden');
    }

    function openEditModal(id) {
      const item = state.transactions.find(function(t) { return t.id === id; });
      if (!item) return;

      document.getElementById('formTxId').value = item.id;
      document.getElementById('modalTitle').innerText = 'Edit Transaksi (' + item.id + ')';
      selectTypeInForm(item.jenis);

      // Convert DD/MM/YYYY to YYYY-MM-DD for date input
      let isoDate = new Date().toISOString().split('T')[0];
      if (item.tanggal && item.tanggal.indexOf('/') !== -1) {
        const p = item.tanggal.split('/');
        if (p.length === 3) isoDate = p[2] + '-' + p[1] + '-' + p[0];
      }
      document.getElementById('formTanggal').value = isoDate;
      document.getElementById('formNominal').value = item.nominal;
      document.getElementById('formDeskripsi').value = item.deskripsi;
      
      // Delay category selection to match populated options
      setTimeout(function() {
        document.getElementById('formKategori').value = item.kategori;
      }, 50);

      document.getElementById('transactionModal').classList.remove('hidden');
    }

    function closeModal() {
      document.getElementById('transactionModal').classList.add('hidden');
    }

    function selectTypeInForm(type) {
      document.getElementById('formJenis').value = type;
      const btnIn = document.getElementById('btnTypeMasuk');
      const btnOut = document.getElementById('btnTypeKeluar');

      if (type === 'Masuk') {
        btnIn.className = 'py-2.5 rounded-xl text-xs font-bold border transition duration-150 flex items-center justify-center space-x-2 bg-emerald-500/20 text-emerald-400 border-emerald-500/50';
        btnOut.className = 'py-2.5 rounded-xl text-xs font-bold border transition duration-150 flex items-center justify-center space-x-2 bg-slate-900 text-slate-400 border-cardborder hover:bg-slate-800';
      } else {
        btnIn.className = 'py-2.5 rounded-xl text-xs font-bold border transition duration-150 flex items-center justify-center space-x-2 bg-slate-900 text-slate-400 border-cardborder hover:bg-slate-800';
        btnOut.className = 'py-2.5 rounded-xl text-xs font-bold border transition duration-150 flex items-center justify-center space-x-2 bg-red-500/20 text-dangerred border-red-500/50';
      }

      populateCategoryDropdown();
    }

    }

    function formatRupiah(number) {
      return new Intl.NumberFormat('id-ID', { style: 'currency', currency: 'IDR', maximumFractionDigits: 0 }).format(number || 0);
    }

    function escapeHtml(text) {
      if (!text) return '';
      return String(text)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#039;');
    }
